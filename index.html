// ==================== VARIABLES GLOBALES ==================== let signer = null; let provider = null; let account = null; let chainId = null; let allTokens = []; // Lista completa de tokens de la API let tokensMap = {}; // Mapa tokenId -> tokenInfo let selectedFromId = null; let selectedToId = null; // ==================== REFERENCIAS DOM ==================== const connectBtn = document.getElementById('connectBtn'); // ==================== FUNCIONES AUXILIARES ==================== function log(msg) { logsDiv.textContent += msg + '\n'; logsDiv.scrollTop = logsDiv.scrollHeight; } function setStatus(msg, isError = false) { statusDiv.textContent = msg; statusDiv.style.background = isError ? OBTENER TOKENS DE LA API ==================== async function fetchTokens() { try { const res = await fetch(`${INTENTS_API}/tokens/${API_KEY}`); if (!res.ok) throw new Error(`Error ${res.status}: ${await res.text()}`); const data = await res.json(); allTokens = data.tokens || []; // Construir mapa: clave = chainId + address (para EVM) allTokens.forEach(t => { const key = t.chainId + ':' + t.address; tokensMap[key] = t; // También guardar por symbol para búsqueda rápida tokensMap[t.symbol] = t; }); log(`✅ ${allTokens.length} tokens cargados desde Intents API`); return allTokens; } catch (e) { log(`❌ Error al cargar tokens: ${e.message}`); setStatus('Error al cargar tokens', true); return []; } } // ==================== 2. CONECTAR WALLET ==================== async function connectWallet() { if (!window.ethereum) { alert('No se detecta una wallet (MetaMask, Trust, etc.)'); return; } try { // Solicitar cuentas const accounts = await window.ethereum.request({ method: 'eth_requestAccounts' }); account = accounts[0]; provider = new ethers.BrowserProvider(window.ethereum); signer = await provider.getSigner(); // Obtener chainId actual const net = await provider.getNetwork(); chainId = Number(net.chainId); // Si no es Tuxachain, preguntar si agregar if (chainId !== TUXA_CHAIN.chainId) { const shouldAdd = confirm('Tu wallet no está en Tuxachain. ¿Quieres agregar la red Tuxachain?'); if (shouldAdd) { await window.ethereum.request({ method: 'wallet_addEthereumChain', params: [{ chainId: '0x' + TUXA_CHAIN.chainId.toString(16), chainName: 'Tuxachain', rpcUrls: [TUXA_CHAIN.rpc], nativeCurrency: TUXA_CHAIN.nativeCurrency, blockExplorerUrls: [TUXA_CHAIN.explorer], }] }); // Esperar cambio de red await new Promise(resolve => setTimeout(resolve, 1000)); const newNet = await provider.getNetwork(); chainId = Number(newNet.chainId); } } accountDisplay.textContent = `👤 ${account.slice(0,6)}...${account.slice(-4)}`; networkDisplay.textContent = `🔗 ${chainId === TUXA_CHAIN.chainId ? 'Tuxachain' : chainId}`; connectBtn.textContent = '✅ Conectado'; connectBtn.disabled = true; swapBtn.disabled = false; log(`🔗 Conectado a ${chainId === TUXA_CHAIN.chainId ? 'Tuxachain' : chainId} - ${account}`); setStatus('Wallet conectada correctamente'); // Actualizar selectores según la red actual updateTokenSelectors(); return true; } catch (e) { log(`❌ Error al conectar: ${e.message}`); setStatus('Error al conectar la wallet', true); return false; } } // ==================== 3. ACTUALIZAR SELECTORES DE TOKENS ==================== function updateTokenSelectors() { // Limpiar selects fromSelect.innerHTML = ''; toSelect.innerHTML = ''; // Filtrar tokens según la red actual let fromTokens = []; let toTokens = []; if (chainId === TUXA_CHAIN.chainId) { // Estamos en Tuxachain: mostrar los tokens definidos (usando las direcciones) const tuxaTokenIds = Object.values(TUXA_TOKENS).map(addr => `evm:${chainId}:${addr}`); fromTokens = allTokens.filter(t => tuxaTokenIds.includes(t.id)); // Si no se encuentran en allTokens, los creamos manualmente if (fromTokens.length === 0) { fromTokens = Object.entries(TUXA_TOKENS).map(([symbol, addr]) => ({ id: `evm:${chainId}:${addr}`, symbol: symbol, name: symbol, chainId: chainId, address: addr, decimals: 18, })); } // Destino: todos los demás tokens (de otras redes) toTokens = allTokens.filter(t => t.chainId !== chainId); } else { // Estamos en otra red EVM (Ethereum, Aurora, etc.) // Mostrar tokens de esa red como origen fromTokens = allTokens.filter(t => Number(t.chainId) === chainId); // Destino: tokens de Tuxachain (y posiblemente otras) toTokens = allTokens.filter(t => Number(t.chainId) === TUXA_CHAIN.chainId); // También permitir otras redes? Podemos mostrar todas excepto la actual // Pero por simplicidad, destino solo Tuxachain. } // Llenar fromSelect fromTokens.forEach(t => { const opt = document.createElement('option'); opt.value = t.id; opt.textContent = `${t.symbol} (${t.name || t.symbol})`; fromSelect.appendChild(opt); }); // Llenar toSelect toTokens.forEach(t => { const opt = document.createElement('option'); opt.value = t.id; opt.textContent = `${t.symbol} (${t.name || t.symbol})`; toSelect.appendChild(opt); }); if (fromTokens.length === 0) log('⚠️ No se encontraron tokens en esta red'); if (toTokens.length === 0) log('⚠️ No se encontraron tokens destino'); } // ==================== 4. OBTENER COTIZACIÓN ==================== async function getQuote(fromId, toId, amount) { const url = `${INTENTS_API}/quote/${API_KEY}`; const body = { swapType: 'EXACT_INPUT', originAsset: fromId, destinationAsset: toId, amount: amount.toString(), recipient: account, recipientType: 'DESTINATION_CHAIN', refundTo: account, refundType: 'ORIGIN_CHAIN', slippageTolerance: 100, // 1% deadline: new Date(Date.now() + 3600000).toISOString(), }; const res = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body), }); if (!res.ok) throw new Error(`Quote error: ${await res.text()}`); return res.json(); } // ==================== 5. OBTENER INTERMEDIARIO ==================== async function getIntermediary(wallet) { const url = `${INTENTS_API}/intermediary/${API_KEY}`; const res = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ wallet: wallet }), }); if (!res.ok) throw new Error(`Intermediary error: ${await res.text()}`); const data = await res.json(); // La respuesta tiene { evm, near, solana } según la red return data.evm; // Para EVM } // ==================== 6. EJECUTAR SWAP ==================== async function executeSwap() { if (!signer) { setStatus('Primero conecta tu wallet', true); return; } const fromId = fromSelect.value; const toId = toSelect.value; const amount = amountInput.value; if (!fromId || !toId || !amount || parseFloat(amount) <= 0) { setStatus('Selecciona tokens y monto válido', true); return; } swapBtn.disabled = true; setStatus('⏳ Procesando...'); log('🚀 Iniciando intercambio...'); try { // 1. Obtener cotización const quote = await getQuote(fromId, toId, ethers.parseUnits(amount, 18)); log(`📊 Cotización recibida: ${quote.estimatedOutput} tokens destino`); // 2. Obtener dirección intermediaria const intermediary = await getIntermediary(account); log(`📍 Intermediario: ${intermediary}`); // 3. Transferir tokens al intermediario (usando ethers) const tokenInfo = allTokens.find(t => t.id === fromId); if (!tokenInfo) throw new Error('Token origen no encontrado en la API'); const tokenAddress = tokenInfo.address; const tokenContract = new ethers.Contract( tokenAddress, ['function transfer(address to, uint256 amount) returns (bool)'], signer ); const amountWei = ethers.parseUnits(amount, tokenInfo.decimals || 18); log(`🔄 Enviando ${amount} ${tokenInfo.symbol} al intermediario...`); const tx = await tokenContract.transfer(intermediary, amountWei); setStatus(`⏳ Transacción enviada, esperando confirmación...`); log(`📝 TX hash: ${tx.hash}`); const receipt = await tx.wait(); log(`✅ Confirmado en bloque ${receipt.blockNumber}`); // 4. Notificar a Intents que el depósito se realizó const settleRes = await fetch(`${INTENTS_API}/settle/${API_KEY}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ quoteId: quote.id, txHash: tx.hash, }), }); if (!settleRes.ok) { const errText = await settleRes.text(); log(`⚠️ Notificación de liquidación falló: ${errText}`); } else { log('✅ Liquidación notificada correctamente'); } setStatus(`✅ ¡Intercambio completado! TX: ${tx.hash.slice(0,10)}...`); } catch (e) { log(`❌ Error: ${e.message}`); setStatus(`Error: ${e.message}`, true); } finally { swapBtn.disabled = false; } } // ==================== 7. EVENTOS Y ARRANQUE ==================== connectBtn.addEventListener('click', connectWallet); swapBtn.addEventListener('click', executeSwap); // Detectar cambio de red if (window.ethereum) { window.ethereum.on('chainChanged', (newChain) => { chainId = parseInt(newChain, 16); networkDisplay.textContent = `🔗 ${chainId === TUXA_CHAIN.chainId ? 'Tuxachain' : chainId}`; updateTokenSelectors(); log(`🔄 Red cambiada a ${chainId}`); }); window.ethereum.on('accountsChanged', (accounts) => { if (accounts.length === 0) { account = null; accountDisplay.textContent = ''; connectBtn.textContent = '🔌 Conectar Wallet'; connectBtn.disabled = false; swapBtn.disabled = true; setStatus('Wallet desconectada', true); log('👋 Wallet desconectada'); } else { account = accounts[0]; accountDisplay.textContent = `👤 ${account.slice(0,6)}...${account.slice(-4)}`; log(`👤 Cuenta cambiada a ${account}`); } }); } // Inicializar (async function init() { await fetchTokens(); // Si ya hay una cuenta conectada (por ejemplo, si la página recarga) if (window.ethereum && window.ethereum.selectedAddress) { await connectWallet(); } else { setStatus('Conecta tu wallet para empezar'); } })();
